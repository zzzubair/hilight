Start here

Hilight is one CLI, one local server, and one browser page. Every code card below is read from this repo when the board opens, and turns Stale if its lines change.

  • Each coloured arrow is one step of a flow; the legend names the flows.
  • Arrows start and end on the exact lines responsible, highlighted in the flow's colour.
  • Draw on the canvas, then Queue marks. Select a card and press Comment to start a thread.
  • Double-click How the canvas is built to go one level deeper.

How the canvas is built

One board file, two views. The server reads cards and flows from the file, places them, and the browser routes every arrow around the cards.

Arrows go around cards

Each step is routed from its source lines to its target lines through the gaps between cards, then overlapping segments of different flows are pushed apart so both stay visible. 

export function createLinkElements(cards, cardElements, flows = [], { scrollOf = () => 0, emphasis = new Set() } = {}) {
  const byCard = new Map(
    cardElements.filter((element) => !element.isDeleted).map((element) => [cardIdOfElement(element), element]),
  );
  const models = new Map(cards.map((card) => [card.id, card]));
  const obstacles = [...byCard.values()].map(elementRect);
  const routes = [];
  const meta = [];
  const seen = new Set();
  for (const step of flowSteps(flows)) {
    const fromElement = byCard.get(step.from.card);
    const toElement = byCard.get(step.to.card);
    if (!fromElement || !toElement || fromElement === toElement) continue;
    const id = `${LINK_ELEMENT_PREFIX}${step.key}`;
    if (seen.has(id)) continue;
    seen.add(id);
    const a = elementRect(fromElement);
    const b = elementRect(toElement);
    const fromY = lineY(fromElement, models.get(step.from.card), step.from.lines, scrollOf(step.from.card));
    const toY = lineY(toElement, models.get(step.to.card), step.to.lines, scrollOf(step.to.card));
    const route =
      routeAround(cardPorts(a, fromY, b), cardPorts(b, toY, a), obstacles) || connectorPoints(a, b, { fromY, toY });
    routes.push(route.map((point) => [...point]));
    meta.push({ id, step, level: models.get(step.from.card)?.level ?? "" });
  }
  nudgeOverlaps(routes);

Stale or moved

Compares the live lines with the pinned hash. If the pinned copy shows up elsewhere in the file the card follows it as Moved; otherwise it is Stale, so the board never silently describes old code. 

export function checkCodeCard(code, file) {
  if (code.error) return { status: "invalid", error: code.error };
  if (file.error) return { status: "missing", error: file.error };
  const lines = file.lines || [];
  const span = code.end - code.start + 1;
  const pinnedSnapshot =
    Array.isArray(code.snapshot) && code.pinned && codeRangeHash(code.snapshot) === code.pinned ? code.snapshot : null;
  if (code.start > lines.length) {
    if (code.pinned && pinnedSnapshot) {
      const at = findBlock(lines, pinnedSnapshot, code.start - 1);
      if (at >= 0) return movedResult(lines, at, span);
    }
    return { status: "missing", error: `line ${code.start} is past the end of the file (${lines.length} lines)` };
  }
  const current = lines.slice(code.start - 1, code.end);
  const live = codeRangeHash(current);
  const total = lines.length;
  if (!code.pinned) return { status: "unpinned", live, lines: current, total };
  if (code.pinned === live) return { status: "fresh", live, lines: current, total };
  if (pinnedSnapshot) {
    const at = findBlock(lines, pinnedSnapshot, code.start - 1);
    if (at >= 0) return movedResult(lines, at, span);
  }
  return {
    status: "stale",
    live,
    lines: current,
    total,
    ...(pinnedSnapshot ? { snapshot: pinnedSnapshot } : {}),
  };
}

CLI starts one detached server

The first hilight command spawns the server and lets it outlive the command; every later command finds it and reuses it, so all review pages share one process. 

async function startServer(port, { alsoListen = [] } = {}) {
  await ensureStateDir();
  const entry = resolveServerEntry();
  let logFd = null;
  try {
    logFd = openSync(serverLogFile(), "a");
  } catch {
    // If logging cannot be initialized, keep the server behavior unchanged.
  }
  try {
    const args = [entry, "server", "--port", String(port)];
    for (const host of alsoListen) args.push("--also-listen", host);
    const child = spawn(process.execPath, args, createServerSpawnOptions(logFd));
    child.unref();
  } finally {
    if (logFd !== null) closeSync(logFd);
  }
}

One locked take drains the batch

Reads every pending prompt under the store's lock and hands them over together, so two polls can never each get half of your feedback. 

  async takeFeedback(key) {
    return this.runExclusive(async () => {
      const state = await this.readState();
      const session = state.sessions[key];
      if (!session) {
        return { status: "missing" };
      }
      // Prompts queued before the session ended (a browser send-and-end) must still reach the
      // agent, so deliver them before reporting the ended state; the next poll then sees ended.
      const prompts = session.prompts || [];
      // Layout warnings stay passive until the user queues them. Only fatal artifact
      // failures can reach the agent without explicit user action.
      const artifactFailures = Array.isArray(session.artifact_failures) ? session.artifact_failures : [];
      const alreadyEnded = session.status === "ended";
      if (prompts.length === 0 && artifactFailures.length === 0) {
        return alreadyEnded ? { status: "ended", ended_by: session.ended_by } : { status: "waiting" };
      }
      // Seen (docs/delivery-acks.md): this take drains every pending prompt and `queuePrompts`
      // shares this lock, so the user entries with no stamp are exactly this batch. The sequence
      // rides on the result so a closed-poll restore can undo only this take's stamps.
      const deliverySeq = normalizeRevision(session.delivery_seq) + 1;
      const result = {
        status: "feedback",
        dom_snapshot: session.dom_snapshot || "",
        prompts,
        ...(artifactFailures.length > 0 ? { artifact_failures: artifactFailures } : {}),
        ...(alreadyEnded ? { session_ended: true, ended_by: session.ended_by } : {}),
        ...(prompts.length > 0 ? { delivery_seq: deliverySeq } : {}),
      };
